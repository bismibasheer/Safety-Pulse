

// app/Http/Controllers/BrandSalesController.php
namespace App\Http\Controllers;

use App\Models\Brand;
use Illuminate\Http\Request;

class BrandSalesController extends Controller
{
    public function showChart()
    {
        // Query to count customer requests per brand
        $brandSales = Brand::withCount('customerRequests') // Use 'customerRequests' relation from Brand
            ->get();

        // Prepare data for the chart
        $brands = $brandSales->pluck('name'); // assuming 'name' is the brand's name field
        $sales = $brandSales->pluck('customer_requests_count'); // 'customer_requests_count' is the count of related customer requests

        return view('brand_sales_chart', compact('brands', 'sales'));
    }
}
